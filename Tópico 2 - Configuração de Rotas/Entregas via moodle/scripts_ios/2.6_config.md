## ATIVIDADE 2.6 - RIPv2 3 ROTEADORES (Triangular)
```bash
! --- ROTEADOR R1 ---
enable
configure terminal
hostname R1
interface FastEthernet0/0
 ip address 223.1.1.3 255.255.255.0
 no shutdown
 exit
interface Serial0/0/0
 ip address 223.1.9.1 255.255.255.0
 no shutdown
 exit
interface Serial0/0/1
 ip address 223.1.7.1 255.255.255.0
 no shutdown
 exit
router rip
 version 2
 network 223.1.1.0
 network 223.1.9.0
 network 223.1.7.0
 no auto-summary
 exit
end
write memory
```
